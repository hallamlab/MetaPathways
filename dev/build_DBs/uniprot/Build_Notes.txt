# SwissProt
wget https://ftp.uniprot.org/pub/databases/uniprot/current_release/knowledgebase/complete/uniprot_sprot.fasta.gz
gunzip uniprot_sprot.fasta.gz
mv uniprot_sprot.fasta swissprot_$(date -u '+%Y-%m-%d').fasta
fastdb -p formatted/swissprot swissprot_$(date -u '+%Y-%m-%d').fasta
grep '>' swissprot_$(date -u '+%Y-%m-%d').fasta > formatted/swissprot-names.txt

# Mapping SwissProt to ECs
wget https://ftp.uniprot.org/pub/databases/uniprot/current_release/knowledgebase/idmapping/idmapping.dat.gz
wget https://ftp.expasy.org/databases/enzyme/enzyme.dat
python swissprot_mapper.py enzyme.dat ./
